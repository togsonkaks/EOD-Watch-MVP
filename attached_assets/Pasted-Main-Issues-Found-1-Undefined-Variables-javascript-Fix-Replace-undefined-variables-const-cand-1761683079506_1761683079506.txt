// Fix: Update mini chart data loading with proper error handling
async function loadMiniChartData(symbol) {
  try {
    const miniChartData = miniCharts[symbol];
    if (!miniChartData) return;
    
    // Use the same fetchBars function for consistency
    const bars = await fetchBars(symbol, 60); // Last 60 days for mini charts
    
    if (bars && bars.length > 0) {
      const chartData = bars.map(bar => ({
        time: bar.time,
        open: bar.open,
        high: bar.high,
        low: bar.low,
        close: bar.close
      }));
      
      miniChartData.candleSeries.setData(chartData);
      
      // Update price info
      const latestBar = bars[bars.length - 1];
      if (latestBar && miniChartData.priceEl) {
        miniChartData.priceEl.textContent = `$${latestBar.close.toFixed(2)}`;
        
        if (bars.length > 1) {
          const prevBar = bars[bars.length - 2];
          const change = latestBar.close - prevBar.close;
          const changePercent = ((change / prevBar.close) * 100).toFixed(2);
          const changeText = `${change >= 0 ? '+' : ''}${change.toFixed(2)} (${changePercent}%)`;
          
          if (miniChartData.changeEl) {
            miniChartData.changeEl.textContent = changeText;
            miniChartData.changeEl.className = `mini-chart-change ${change >= 0 ? 'positive' : 'negative'}`;
          }
        }
      }
      
      miniChartData.loadingDiv.style.display = 'none';
    }
  } catch (error) {
    console.error(`Error loading mini chart data for ${symbol}:`, error);
    if (miniChartData && miniChartData.loadingDiv) {
      miniChartData.loadingDiv.textContent = 'Error';
    }
  }
}

// Enhanced mini chart disposal
function disposeMiniCharts() {
  Object.keys(miniCharts).forEach(symbol => {
    const mini = miniCharts[symbol];
    if (mini && mini.chart) {
      try {
        mini.chart.remove();
      } catch (e) {
        console.warn('Error removing mini chart:', e);
      }
    }
  });
  miniCharts = {};
}
