## 🧱 How It Works Technically

* You’ll use a **mouse event listener** (`mousedown`, `mousemove`, `mouseup`) on the overlay canvas.
* While moving the cursor between clicks, you draw a *temporary* line (`ctx.setLineDash([4, 2])` to make it dotted).
* On second click, that line becomes **permanent**.
* For moving/extending:

  * Detect if the cursor is within ~5–8px of an endpoint.
  * On drag, update that endpoint’s coordinates.

## 🔧 Here’s the Upgraded Script (Core Logic Only)

This version gives you:

* Click-to-start and click-to-end drawing
* **Hover guide line** before placement
* **Draggable lines** after creation

```html
<script>
const chart = LightweightCharts.createChart(document.getElementById('chart'), { width: 800, height: 500 });
const candleSeries = chart.addCandlestickSeries();
candleSeries.setData([
  { time: '2024-10-20', open: 107000, high: 107500, low: 106800, close: 107300 },
  { time: '2024-10-21', open: 107300, high: 108000, low: 107000, close: 107800 },
  { time: '2024-10-22', open: 107800, high: 108200, low: 107500, close: 108100 },
]);

const canvas = document.createElement('canvas');
canvas.classList.add('overlay');
document.getElementById('chart').appendChild(canvas);
const ctx = canvas.getContext('2d');
let lines = [];
let currentLine = null;
let dragging = null;

function resize() {
  canvas.width = chart.chartElement().clientWidth;
  canvas.height = chart.chartElement().clientHeight;
}
resize();
window.addEventListener('resize', resize);

function drawAll() {
  ctx.clearRect(0,0,canvas.width,canvas.height);
  for (const line of lines) {
    ctx.beginPath();
    ctx.moveTo(line.a.x, line.a.y);
    ctx.lineTo(line.b.x, line.b.y);
    ctx.strokeStyle = 'rgba(255,255,255,0.9)';
    ctx.lineWidth = 1.5;
    ctx.stroke();
  }
  if (currentLine) {
    ctx.setLineDash([5,3]);
    ctx.beginPath();
    ctx.moveTo(currentLine.a.x, currentLine.a.y);
    ctx.lineTo(currentLine.b.x, currentLine.b.y);
    ctx.strokeStyle = 'rgba(255,255,255,0.5)';
    ctx.lineWidth = 1;
    ctx.stroke();
    ctx.setLineDash([]);
  }
}

canvas.addEventListener('mousedown', e=>{
  const {offsetX:x, offsetY:y} = e;
  // check if near endpoint for dragging
  for (const line of lines) {
    if (Math.hypot(line.a.x-x, line.a.y-y)<6) { dragging={line,point:'a'}; return; }
    if (Math.hypot(line.b.x-x, line.b.y-y)<6) { dragging={line,point:'b'}; return; }
  }
  if (!currentLine) currentLine={a:{x,y}, b:{x,y}};
  else { lines.push(currentLine); currentLine=null; }
  drawAll();
});

canvas.addEventListener('mousemove', e=>{
  const {offsetX:x, offsetY:y} = e;
  if (dragging) {
    dragging.line[dragging.point] = {x,y};
  } else if (currentLine) {
    currentLine.b = {x,y};
  }
  drawAll();
});

canvas.addEventListener('mouseup', ()=>{ dragging=null; });
</script>
```
