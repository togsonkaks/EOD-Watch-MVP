## ⚙️ Step-by-Step: Adding Custom Trendline Drawing to Lightweight Charts

### 🧩 1️⃣ Set Up Your Chart

```html
<!DOCTYPE html>
<html>
<head>
  <script src="https://unpkg.com/lightweight-charts/dist/lightweight-charts.standalone.production.js"></script>
  <style>
    #container {
      position: relative;
      width: 100%;
      height: 500px;
    }
    canvas.overlay {
      position: absolute;
      top: 0;
      left: 0;
      pointer-events: all; /* important — lets you click on top */
    }
  </style>
</head>
<body>
  <div id="container"></div>
  <script>
    // Create the chart
    const chart = LightweightCharts.createChart(document.getElementById('container'), {
      layout: { background: { color: '#0d1117' }, textColor: '#d1d4dc' },
      grid: { vertLines: { color: '#222' }, horzLines: { color: '#222' } },
    });
    const series = chart.addCandlestickSeries();

    // Add some data
    series.setData([
      { time: '2025-10-23', open: 107000, high: 108000, low: 106500, close: 107800 },
      { time: '2025-10-24', open: 107800, high: 108400, low: 107200, close: 108200 },
      { time: '2025-10-25', open: 108200, high: 108500, low: 107900, close: 108400 },
      { time: '2025-10-26', open: 108400, high: 109000, low: 108000, close: 108800 },
    ]);
```

### ⚙️ 2️⃣ Add the Overlay Canvas

This sits *above* the chart and tracks mouse clicks.

```js
    const container = document.getElementById('container');
    const canvas = document.createElement('canvas');
    canvas.classList.add('overlay');
    container.appendChild(canvas);
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = container.clientWidth;
      canvas.height = container.clientHeight;
    }
    resizeCanvas();
    new ResizeObserver(resizeCanvas).observe(container);
```

---

### ⚙️ 3️⃣ Sync Trendline Coordinates With Chart Scale

When you click on the chart, you get pixel coordinates.
But you can **convert those to price/time** (so your lines move correctly when zooming/panning).

```js
    const lines = [];
    let drawingLine = null;
    let dragging = null;

    function drawAll() {
      ctx.clearRect(0,0,canvas.width,canvas.height);
      ctx.lineWidth = 1.5;
      ctx.strokeStyle = 'rgba(255,255,255,0.9)';
      for (const l of lines) {
        const a = {
          x: chart.timeScale().timeToCoordinate(l.aTime),
          y: series.priceToCoordinate(l.aPrice),
        };
        const b = {
          x: chart.timeScale().timeToCoordinate(l.bTime),
          y: series.priceToCoordinate(l.bPrice),
        };
        if (a.x && b.x) {
          ctx.beginPath();
          ctx.moveTo(a.x, a.y);
          ctx.lineTo(b.x, b.y);
          ctx.stroke();
        }
      }
      // preview line
      if (drawingLine) {
        ctx.setLineDash([5,3]);
        ctx.beginPath();
        ctx.moveTo(drawingLine.start.x, drawingLine.start.y);
        ctx.lineTo(drawingLine.end.x, drawingLine.end.y);
        ctx.strokeStyle = 'rgba(255,255,255,0.5)';
        ctx.stroke();
        ctx.setLineDash([]);
      }
    }

    chart.subscribeCrosshairMove(drawAll);
```

### ⚙️ 4️⃣ Add Click + Move Interactions

This handles both preview and permanent lines.

```js
    canvas.addEventListener('mousedown', e=>{
      const rect = canvas.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;

      if (!drawingLine) {
        drawingLine = { start: {x,y}, end: {x,y} };
      } else {
        const aTime = chart.timeScale().coordinateToTime(drawingLine.start.x);
        const bTime = chart.timeScale().coordinateToTime(x);
        const aPrice = series.coordinateToPrice(drawingLine.start.y);
        const bPrice = series.coordinateToPrice(y);
        lines.push({ aTime, bTime, aPrice, bPrice });
        drawingLine = null;
      }
      drawAll();
    });

    canvas.addEventListener('mousemove', e=>{
      if (drawingLine) {
        const rect = canvas.getBoundingClientRect();
        drawingLine.end = { x: e.clientX - rect.left, y: e.clientY - rect.top };
        drawAll();
      }
    });
```

## 💡 TL;DR

To “plug it in”:

1. Create your chart with Lightweight Charts.
2. Add an overlay `<canvas>` directly above it.
3. Draw and store lines using `coordinateToTime()` + `priceToCoordinate()` conversions.
4. On zoom/pan, just re-render (`drawAll()`) so lines sync with the chart.
